```typescript
// oracleService.ts - AFTER

async getPriceWithFallback(mint: string): Promise<{ 
  price: number; 
  source: string;
  confidence: number;
  timestamp: number;
}> {
  // Check cache first (30s TTL)
  const cached = priceCache.get(mint);
  if (cached && Date.now() - cached.timestamp < CACHE_TTL) {
    return {
      price: cached.price,
      source: cached.source,
      confidence: 0.95,
      timestamp: cached.timestamp,
    };
  }

  // Define 5 price sources in priority order
  const sources: PriceSource[] = [
    { name: 'Jupiter', priority: 1, fetchFn: this.fetchFromJupiter.bind(this) },
    { name: 'Pyth', priority: 2, fetchFn: this.fetchFromPyth.bind(this) },
    { name: 'Switchboard', priority: 3, fetchFn: this.fetchFromSwitchboard.bind(this) },
    { name: 'DexScreener', priority: 4, fetchFn: this.fetchFromDexScreener.bind(this) },
    { name: 'CoinGecko', priority: 5, fetchFn: this.fetchFromCoinGecko.bind(this) },
  ];

  // Try top 3 sources in parallel
  const results = await Promise.allSettled(
    topSources.map(async (source) => {
      // Check circuit breaker before calling
      if (!this.isCircuitBreakerClosed(source.name)) {
        return null; // Skip failed sources
      }
      
      try {
        const price = await source.fetchFn(mint);
        if (price !== null) {
          this.recordSuccess(source.name);
          return { source: source.name, price };
        }
        return null;
      } catch (error: any) {
        this.recordFailure(source.name);
        return null;
      }
    })
  );

  // Collect all successful prices
  const prices = results.filter(r => r.status === 'fulfilled' && r.value);
  
  if (prices.length === 0) {
    throw new Error('All oracle sources failed');
  }

  // Validate consensus and detect deviation
  const validatedPrice = this.validatePriceConsensus(prices);

  // Cache result
  priceCache.set(mint, {
    price: validatedPrice.price,
    timestamp: Date.now(),
    source: validatedPrice.source,
  });

  return {
    price: validatedPrice.price,
    source: validatedPrice.source,
    confidence: this.calculateConfidence(prices.length),
    timestamp: Date.now(),
  };
}
```

### **2. Deviation Detection System** 🎯

```typescript
validatePriceConsensus(prices: { source: string; price: number }[]) {
  if (prices.length === 1) {
    return prices[0]; // No validation possible
  }
  
  // Calculate median price
  const sortedPrices = [...prices].sort((a, b) => a.price - b.price);
  const medianPrice = sortedPrices[medianIndex].price;
  
  // Find prices within 5% deviation threshold
  const validPrices = prices.filter(p => {
    const deviation = Math.abs((p.price - medianPrice) / medianPrice);
    return deviation <= PRICE_DEVIATION_THRESHOLD; // 0.05 = 5%
  });
  
  if (validPrices.length === 0) {
    console.warn('[Oracle] Price deviation too high, using median');
    return sortedPrices[medianIndex]; // Use median as fallback
  }
  
  // Use highest priority source from valid prices
  const priorityOrder = ['Jupiter', 'Pyth', 'Switchboard', 'DexScreener', 'CoinGecko'];
  const validSorted = validPrices.sort((a, b) => {
    return priorityOrder.indexOf(a.source) - priorityOrder.indexOf(b.source);
  });
  
  return validSorted[0];
}
```

**Deviation Scenarios:**

| Scenario | Prices | Median | Valid (≤5%) | Action |
|----------|--------|--------|-------------|--------|
| **Normal** | J: $150, P: $151, S: $149 | $150 | All 3 | Use Jupiter ($150) |
| **Moderate Deviation** | J: $150, P: $158, S: $151 | $151 | J, S | Use Jupiter ($150) |

### **3. Circuit Breaker Pattern** 🔌

**States:**
- **CLOSED** ✅ - Normal operation, allowing requests
- **OPEN** 🔴 - Too many failures, blocking requests
- **HALF_OPEN** 🟡 - Testing after timeout

**Configuration:**
```typescript
CIRCUIT_BREAKER_THRESHOLD = 3;      // Failures before opening
CIRCUIT_BREAKER_TIMEOUT = 30000;    // ms before half-open
```

**State Transitions:**
```
CLOSED ──[3 failures]──> OPEN
OPEN ──[30s timeout]──> HALF_OPEN
HALF_OPEN ──[1 success]──> CLOSED
HALF_OPEN ──[1 failure]──> OPEN
```

**Example:**
```typescript
// Record failure
recordFailure('Jupiter'):
  failures = 1 → 2 → 3
  state = CLOSED → OPEN
  console.warn('[Oracle] Circuit breaker OPEN for Jupiter after 3 failures')

// After 30 seconds
isCircuitBreakerClosed('Jupiter'):
  state = OPEN, lastFailureTime = 31s ago
  → Transition to HALF_OPEN
  → Allow test request
```

```typescript
calculateConfidence(sourceCount: number): number {
  const confidenceMap: Record<number, number> = {
    1: 0.60, // Single source - low confidence
    2: 0.75, // Two sources agree - medium
    3: 0.90, // Three sources agree - high
    4: 0.95, // Four sources agree - very high
    5: 0.99, // All five agree - maximum
  };
  return confidenceMap[sourceCount] || 0.60;
}
```
