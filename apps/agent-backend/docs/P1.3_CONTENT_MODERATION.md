### **2. Confidence Scoring System** 📊

**Scoring Algorithm:**
```typescript
// Overall Score Calculation:
overallScore = (
  toxicity.score * 0.4 +      // 40% weight
  scam.score * 0.4 +          // 40% weight
  spam.score * 0.2            // 20% weight
);

// Confidence = 100 - overallScore
confidence = Math.max(0, 100 - overallScore);

// Recommendation Logic:
if (overallScore >= 80 || autoRejectToxic || autoRejectScam) {
  recommendation = 'REJECT';
} else if (overallScore >= 40 || requireHumanReview) {
  recommendation = 'REVIEW';
} else {
  recommendation = 'APPROVE';
}
```

### **Example 4: Tiered Moderation System**

```typescript
// Different rules for different user tiers
function getModerationConfig(userTier: string) {
  switch (userTier) {
    case 'NEW_USER':
      return {
        strictMode: true,
        autoRejectToxic: true,
        autoRejectScam: true,
        requireHumanReview: true,
      };
      
    case 'TRUSTED_USER':
      return {
        autoRejectToxic: true,
        autoRejectScam: true,
        requireHumanReview: false,
      };
      
    case 'MODERATOR':
      return {
        autoRejectToxic: false,
        autoRejectScam: false,
        requireHumanReview: false,
      };
      
    default:
      return {
        autoRejectToxic: true,
        autoRejectScam: true,
      };
  }
}

app.use('/api/posts/*', (c, next) => {
  const userTier = c.get('userTier');
  const config = getModerationConfig(userTier);
  return contentModerator(config)(c, next);
});
```
