function generateNonce(): string {
  const crypto = require('crypto');
  return crypto.randomBytes(32).toString('hex');
  // → "a1b2c3d4e5f6..." (64 hex chars = 32 bytes)
}

**Signature Verification:**
```typescript
async function verifySignature(
  message: string,
  signature: string,
  publicKey: string
): Promise<boolean> {
  const { verify } = await import('@noble/ed25519');
  
  const messageBytes = new TextEncoder().encode(message);
  const signatureBytes = Buffer.from(signature, 'base64');
  const publicKeyBytes = new PublicKey(publicKey).toBytes();
  
  return await verify(signatureBytes, messageBytes, publicKeyBytes);
}
```

**Ed25519 Properties:**
- ✅ Strong cryptographic security

  // Remove expired challenges
  for (const [key, challenge] of cache.challenges.entries()) {
    if (now > challenge.expiresAt) {
      cache.challenges.delete(key);

### **Example 2: Frontend Integration (React)**

```typescript
// hooks/useWalletVerification.ts
import { useState } from 'react';
import { useWallet } from '@solana/wallet-adapter-react';

export function useWalletVerification() {
  const { wallet, signMessage } = useWallet();
  const [isVerified, setIsVerified] = useState(false);
  
  const verifyWallet = async () => {
    try {
      // Step 1: Get challenge
      const challengeRes = await fetch('/api/auth/challenge', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          walletAddress: wallet.publicKey.toString(),
        }),
      });
      
      const { challenge } = await challengeRes.json();
      
      // Step 2: Sign message
      const messageBytes = new TextEncoder().encode(challenge.message);
      const signature = await signMessage(messageBytes);
      const signatureBase64 = Buffer.from(signature).toString('base64');
      
      // Step 3: Verify signature
      const verifyRes = await fetch('/api/auth/verify', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          challengeKey: challenge.challenge,
          signature: signatureBase64,
        }),
      });
      
      const result = await verifyRes.json();
      
      if (result.status === 'success') {
        setIsVerified(true);
        
        // Store verification state
        localStorage.setItem('walletVerified', 'true');
        localStorage.setItem('walletAddress', wallet.publicKey.toString());
        
        return { success: true };
      }
      
      return { success: false, error: result.message };
      
    } catch (error) {
      console.error('Verification failed:', error);
      return { success: false, error: 'Verification failed' };
    }
  };
  
  return { isVerified, verifyWallet };
}

// Usage in component
function CreateAgentButton() {
  const { isVerified, verifyWallet } = useWalletVerification();
  
  const handleCreateAgent = async () => {
    if (!isVerified) {
      await verifyWallet();
      return;
    }
    
    // Now verified, can create agent
    const res = await fetch('/api/agent/create', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-Wallet-Address': wallet.publicKey.toString(),
      },
    });
    
    const data = await res.json();
    console.log('Agent created:', data);
  };
  
  return (
    <button onClick={handleCreateAgent}>
      {isVerified ? 'Create Agent' : 'Verify Wallet First'}
    </button>
  );
}
```

## 🛡️ SECURITY PROPERTIES

### **What Makes This Secure:**

1. **Cryptographic Proof**
   ```
   User claims: "I own wallet So11..."
   System challenges: "Sign this message"
   User signs: Ed25519(message, privateKey)
   System verifies: Ed25519_verify(signature, message, publicKey)
   
   Mathematical proof: Only private key holder can create valid signature
   ```

2. **Anti-Replay Protection**
   ```typescript
   // Each challenge is unique
   challenge = hash(wallet + nonce + timestamp)
   
   // One-time use
   After verification → challenge deleted
   
   // Time-limited
   Expires after 5 minutes
   ```

3. **Session Security**
   ```typescript
   // Short-lived sessions (10 minutes)
   verification.expiresAt = verifiedAt + 10*60*1000
   
   // Auto-cleanup
   Expired verifications removed every 5 minutes
   
   // No persistent storage
   In-memory cache only (no database leaks)
   ```

4. **Man-in-the-Middle Protection**
   ```
   Even if attacker intercepts:
   - Challenge message: Useless without private key
   - Signature: Only valid for specific challenge
   - Wallet address: Public information anyway
   
   Attacker cannot:
   - Forge signatures (no private key)
   - Reuse signatures (one-time challenges)
   - Extend sessions (fixed expiry)
   ```
