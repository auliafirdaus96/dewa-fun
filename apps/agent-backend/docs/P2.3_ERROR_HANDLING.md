**Configuration:**
```typescript
interface RetryConfig {
  maxRetries?: number;              // Default: 3
  initialDelayMs?: number;          // Default: 1000ms

  retryableStatusCodes?: number[];  // [408, 429, 500, 502, 503, 504]
```

**Delay Calculation:**
```
Attempt 1: 1000ms
Attempt 2: 1000 × 2 = 2000ms (+ jitter)
Attempt 3: 2000 × 2 = 4000ms (+ jitter)
```

**Usage:**

### **3. Circuit Breaker Pattern** 🔌

**States:**
```
CLOSED → Normal operation, requests flow through
   ↓ (failures >= threshold)
OPEN → Requests blocked, auto-recover after timeout
   ↓ (timeout expires)
HALF_OPEN → Test with limited requests
   ↓ (successes >= threshold) OR (failure)
CLOSED ← Success | OPEN ← Failure
```

**Configuration:**
```typescript
interface CircuitBreakerConfig {
  failureThreshold?: number;     // Default: 5 failures
  successThreshold?: number;     // Default: 3 successes
  timeoutMs?: number;            // Default: 60000ms (1 min)
  halfOpenMaxRequests?: number;  // Default: 3 test requests
}
```

**Usage:**
```typescript
// Protect database calls
const users = await withCircuitBreaker(
  async () => await db.query('SELECT * FROM users'),
  'database-users-breaker',
  { failureThreshold: 5, timeoutMs: 60000 }
);

// When circuit OPENS:
// throw new ServiceUnavailableError(
//   'Service temporarily unavailable (circuit breaker OPEN)',
//   { breakerId: 'database-users-breaker', retryAfter: timestamp }
// );

// Check circuit state
const state = getCircuitBreakerState('database-users-breaker');
console.log(state);
// { state: 'OPEN', failures: 5, nextAttemptTime: 1711824000000 }

// Reset circuit breaker manually
resetCircuitBreaker('database-users-breaker');
```

```typescript
import { withRetry, withCircuitBreaker } from './utils/errorHandler.js';

async function fetchUserData(userId: string) {
  // Combine retry + circuit breaker for maximum resilience
  return await withRetry(
    async () => await withCircuitBreaker(
      async () => {
        const response = await fetch(`https://api.example.com/users/${userId}`);
        
        if (!response.ok) {
          throw new ApiError(`API returned ${response.status}`);
        }

      },
      'user-api-breaker',
      { failureThreshold: 5, timeoutMs: 60000 }
    ),
    { maxRetries: 3, initialDelayMs: 1000 },
    'fetch_user_data'
  );
}

// Usage

  const user = await fetchUserData('user-123');
  console.log('User fetched:', user);
} catch (error) {
  // After 3 retries AND circuit breaker open
  console.error('Failed to fetch user:', error);
}
```

```typescript
async function getUserProfile(userId: string) {

      () => withCircuitBreaker(() => fetchStats(userId), 'stats-breaker'),

    }),

    }),
  ];
  
  const [profile, stats, preferences] = await Promise.all(profilePromises);
  
  return {
    ...profile,
    stats,
    preferences,
  };
}
```
