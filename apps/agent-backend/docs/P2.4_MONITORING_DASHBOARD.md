### **3. Alert System** 🚨

**Alert Conditions:**
```typescript
service.addAlert({
  name: 'high_cpu',
  metric: 'cpu_usage',
  condition: 'gt',        // gt | lt | eq | gte | lte
  threshold: 80,          // Alert when > 80%
  duration: 0,            // seconds (0 = immediate)
  severity: 'warning',    // critical | warning | info
  channels: ['log'],      // email | slack | webhook | log
  message: 'CPU usage is too high',
});
```

**Alert Lifecycle:**
```
Normal → Condition Met → FIRING → Condition Clears → RESOLVED

Example:
CPU: 75% → Normal
CPU: 85% → FIRING (alert sent)
CPU: 70% → RESOLVED (notification sent)
```

import { monitoringService } from './services/monitoringService.js';

// Setup alerts
monitoringService.addAlert({
  name: 'api_slow',
  metric: 'api_response_time',
  condition: 'gt',
  threshold: 500,
  severity: 'warning',
  channels: ['log'],
  message: 'API response time exceeds 500ms',
});

// Middleware to monitor all requests
app.use('*', async (c, next) => {
  const start = Date.now();
  
  await next();
  
  const duration = Date.now() - start;
  
  // Record metrics
  monitoringService.histogram('api_response_time', duration);
  monitoringService.increment('total_requests');
  
  if (c.res.status >= 400) {
    monitoringService.increment('error_count');
  }
});

// Health check
setInterval(async () => {
  await monitoringService.healthCheck('api', async () => {
    const start = Date.now();
    try {
      await fetch('http://localhost:3000/health');
      return {
        status: 'healthy',
        latency: Date.now() - start,
      };
    } catch (error) {
      return {
        status: 'unhealthy',
        latency: Date.now() - start,
        error: error.message,
      };
    }
  });
}, 5000);

// Get dashboard
app.get('/dashboard', (c) => {
  return c.json(monitoringService.getDashboardData());
});

### **Example 2: Database Pool Monitoring**

```typescript
// Monitor database connection pool
async function monitorDatabase() {
  const pool = db.getPool();
  
  await monitoringService.healthCheck('database', async () => {
    const start = Date.now();
    
    try {
      const connections = await pool.getActiveConnections();
      const maxConnections = pool.getMaxConnections();
      const usagePercent = (connections / maxConnections) * 100;
      
      // Record metrics
      monitoringService.metric('db_connections_used', connections);
      monitoringService.metric('db_connections_percent', usagePercent);
      
      // Query performance
      const result = await pool.query('SELECT 1');
      const queryLatency = Date.now() - start;
      monitoringService.histogram('db_query_time', queryLatency);
      
      return {
        status: usagePercent > 90 ? 'degraded' : 'healthy',
        latency: queryLatency,
        details: {
          connections,
          maxConnections,
          usagePercent,
        },
      };
    } catch (error) {
      return {
        status: 'unhealthy',
        latency: Date.now() - start,
        error: error.message,
      };
    }
  });
}

// Alert for pool exhaustion
monitoringService.addAlert({
  name: 'pool_exhaustion',
  metric: 'db_connections_percent',
  condition: 'gt',
  threshold: 90,
  severity: 'critical',
  channels: ['log', 'slack'],
  message: 'Database connection pool nearly exhausted!',
});

// Run every 10 seconds
setInterval(monitorDatabase, 10000);
```

### **Example 4: External Service Monitoring**

```typescript
// Monitor external dependencies
async function monitorExternalServices() {
  const services = [
    { name: 'pyth', url: 'https://hermes.pyth.network/health' },
    { name: 'chainlink', url: 'https://chain.link/health' },
    { name: 'bags_fm', url: 'https://api.bags.fm/health' },
  ];
  
  for (const service of services) {
    await monitoringService.healthCheck(service.name, async () => {
      const start = Date.now();
      
      try {
        const response = await fetch(service.url, { timeout: 5000 });
        const latency = Date.now() - start;
        
        if (!response.ok) {
          throw new Error(`HTTP ${response.status}`);
        }
        
        return {
          status: 'healthy',
          latency,
          details: { status_code: response.status },
        };
      } catch (error) {
        return {
          status: 'unhealthy',
          latency: Date.now() - start,
          error: error.message,
        };
      }
    });
  }
}

// Alert for external service failures
monitoringService.addAlert({
  name: 'pyth_down',
  metric: 'pyth_health',
  condition: 'eq',
  threshold: 0,
  severity: 'critical',
  channels: ['log', 'slack'],
  message: 'Pyth Network is unavailable!',
});

// Check every 30 seconds
setInterval(monitorExternalServices, 30000);
```
